const saveDepartment = async () => {
    if (!canManageDepartments) {
      return;
    }

    const name =
      departmentForm.name.trim();

    const code =
      departmentForm.code
        .trim()
        .toUpperCase();

    const description =
      departmentForm.description.trim();

    if (!name || !code) {
      setDepartmentMutationError(
        "Department name and code are required."
      );

      return;
    }

    setDepartmentSaving(true);
    setDepartmentMutationError("");

    try {
      const response =
        departmentEditingId === null
          ? await createDepartment({
              name,
              code,
              description,
            })
          : await updateDepartment(
              departmentEditingId,
              {
                name,
                code,
                description,
              }
            );

      const data =
        await response.json();

      if (!response.ok || !data.success) {
        throw new Error(
          data.error ||
            "Unable to save department."
        );
      }

      await refreshDepartments();

      setDepartmentModalOpen(false);
      setDepartmentMutationError("");
    } catch (error) {
      setDepartmentMutationError(
        error instanceof Error
          ? error.message
          : "Unable to save department."
      );
    } finally {
      setDepartmentSaving(false);
    }
  };

const deleteDepartment = async (
    department: Department
  ): Promise<boolean> => {
    if (!canManageDepartments) {
      return false;
    }

    const confirmed =
      window.confirm(
        `Delete the "${department.name}" department? Existing metadata and documents will remain, but their department assignment will become unassigned.`
      );

    if (!confirmed) {
      return false;
    }

    setDepartmentMutationError("");

    try {
      const response =
        await deleteDepartmentRequest(
          department.id
        );

      const data =
        await response.json();

      if (!response.ok || !data.success) {
        throw new Error(
          data.error ||
            "Unable to delete department."
        );
      }

      if (
        selectedDepartmentId ===
        department.id
      ) {
        setSelectedDepartmentId(null);
      }

      await refreshDepartments();

      return true;
    } catch (error) {
      setDepartmentMutationError(
        error instanceof Error
          ? error.message
          : "Unable to delete department."
      );

      return false;
    }
  };

const loadDomainRelationshipDefinitions =
    async () => {
      if (!authUser) {
        return;
      }

      try {
        setDomainRelationshipsLoading(true);
        setDomainRelationshipsError("");

        const [
          domainsResponse,
          relationshipTypesResponse,
        ] = await Promise.all([
          getDomainRegistry(),
          getRelationshipTypes(),
        ]);

        const [
          domainsData,
          relationshipTypesData,
        ] = await Promise.all([
          domainsResponse.json(),
          relationshipTypesResponse.json(),
        ]);

        if (
          !domainsResponse.ok ||
          !domainsData.success
        ) {
          throw new Error(
            domainsData.error ||
              "Unable to load the domain registry."
          );
        }

        if (
          !relationshipTypesResponse.ok ||
          !relationshipTypesData.success
        ) {
          throw new Error(
            relationshipTypesData.error ||
              "Unable to load domain relationship types."
          );
        }

        const rawDomains =
          Array.isArray(domainsData.domains)
            ? domainsData.domains
            : [];

        const normalizedDomains =
          rawDomains
            .map(
              (
                domain: Record<string, unknown>
              ) => {
                const id = Number(
                  domain.id
                );

                const domainKey =
                  String(
                    domain.domain_key ??
                      domain.domainKey ??
                      ""
                  ).trim();

                const displayName =
                  String(
                    domain.display_name ??
                      domain.displayName ??
                      domainKey
                  ).trim();

                if (
                  !Number.isInteger(id) ||
                  id <= 0 ||
                  !domainKey
                ) {
                  return null;
                }

                return {
                  id,
                  domain_key: domainKey,
                  display_name:
                    displayName,
                  description:
                    domain.description == null
                      ? null
                      : String(
                          domain.description
                        ),
                  domain_family:
                    domain.domain_family == null
                      ? null
                      : String(
                          domain.domain_family
                        ),
                };
              }
            )
            .filter(
              (domain: unknown): domain is DomainRelationshipDomain =>
                domain !== null
            );

        const rawTypes =
          Array.isArray(
            relationshipTypesData.relationshipTypes
          )
            ? relationshipTypesData.relationshipTypes
            : Array.isArray(
                relationshipTypesData.types
              )
              ? relationshipTypesData.types
              : [];

        const normalizedTypes =
          rawTypes
            .map(
              (
                relationshipType: unknown
              ) => {
                if (
                  typeof relationshipType ===
                  "string"
                ) {
                  return {
                    value:
                      relationshipType,
                    label:
                      relationshipType,
                  };
                }

                if (
                  !relationshipType ||
                  typeof relationshipType !==
                    "object"
                ) {
                  return null;
                }

                const item =
                  relationshipType as Record<
                    string,
                    unknown
                  >;

                const value =
                  String(
                    item.value ??
                      item.relationship_type ??
                      item.relationshipType ??
                      item.key ??
                      ""
                  ).trim();

                if (!value) {
                  return null;
                }

                return {
                  value,
                  label:
                    String(
                      item.label ??
                        item.display_name ??
                        item.displayName ??
                        value
                    ).trim() || value,
                };
              }
            )
            .filter(
              (relationshipType: unknown): relationshipType is DomainRelationshipType =>
                relationshipType !== null
            );

        setDomainRelationshipDomains(
          normalizedDomains
        );

        setDomainRelationshipTypes(
          normalizedTypes
        );
      } catch (error) {
        console.error(
          "Unable to load domain relationship definitions:",
          error
        );

        setDomainRelationshipsError(
          error instanceof Error
            ? error.message
            : "Unable to load domain relationships."
        );
      } finally {
        setDomainRelationshipsLoading(
          false
        );
      }
    };

const extractRelationshipSelections =
    (
      data: Record<string, unknown>
    ): DomainRelationshipSelection[] => {
      const relationships =
        Array.isArray(
          data.relationships
        )
          ? data.relationships
          : [];

      return relationships
        .map(
          (
            relationship: unknown
          ) => {
            if (
              !relationship ||
              typeof relationship !==
                "object"
            ) {
              return null;
            }

            const item =
              relationship as Record<
                string,
                unknown
              >;

            const domainId = Number(
              item.domain_id ??
                item.domainId ??
                (
                  item.domain &&
                  typeof item.domain ===
                    "object"
                    ? (
                        item.domain as Record<
                          string,
                          unknown
                        >
                      ).id
                    : undefined
                )
            );

            const relationshipType =
              String(
                item.relationship_type ??
                  item.relationshipType ??
                  ""
              ).trim();

            if (
              !Number.isInteger(
                domainId
              ) ||
              domainId <= 0 ||
              !relationshipType
            ) {
              return null;
            }

            return {
              domainId,
              relationshipType,
            };
          }
        )
        .filter(
          (
            relationship
          ): relationship is DomainRelationshipSelection =>
            relationship !== null
        );
    };

const saveMetadataDomainRelationships =
    async (
      metadataRecordId: string | number,
      desiredRelationships:
        DomainRelationshipSelection[]
    ) => {
      const numericMetadataId =
        Number(metadataRecordId);

      if (
        !Number.isInteger(
          numericMetadataId
        ) ||
        numericMetadataId <= 0
      ) {
        throw new Error(
          "A valid metadata record is required before domain relationships can be saved."
        );
      }

      setDomainRelationshipsLoading(
        true
      );
      setDomainRelationshipsError("");

      try {
        const currentResponse =
          await getMetadataDomainRelationships(
            numericMetadataId
          );

        const currentData =
          await currentResponse.json();

        if (
          !currentResponse.ok ||
          !currentData.success
        ) {
          throw new Error(
            currentData.error ||
              "Unable to retrieve the current metadata domain relationships."
          );
        }

        const currentRelationships =
          Array.isArray(
            currentData.relationships
          )
            ? currentData.relationships
            : [];

        const currentSelections =
          extractRelationshipSelections(
            currentData
          );

        const keyFor =
          (
            relationship: DomainRelationshipSelection
          ) =>
            `${relationship.domainId}:${relationship.relationshipType}`;

        const desiredKeys =
          new Set(
            desiredRelationships.map(
              keyFor
            )
          );

        const currentKeys =
          new Set(
            currentSelections.map(
              keyFor
            )
          );

        /*
         * Delete relationships that were removed
         * from the selector.
         */
        for (
          const relationship of currentRelationships
        ) {
          if (
            !relationship ||
            typeof relationship !==
              "object"
          ) {
            continue;
          }

          const item =
            relationship as Record<
              string,
              unknown
            >;

          const relationshipId =
            Number(
              item.id
            );

          const domainId =
            Number(
              item.domain_id ??
                item.domainId
            );

          const relationshipType =
            String(
              item.relationship_type ??
                item.relationshipType ??
                ""
            ).trim();

          if (
            !Number.isInteger(
              relationshipId
            ) ||
            relationshipId <= 0 ||
            !Number.isInteger(
              domainId
            ) ||
            domainId <= 0 ||
            !relationshipType
          ) {
            continue;
          }

          const key =
            `${domainId}:${relationshipType}`;

          if (
            !desiredKeys.has(key)
          ) {
            const response =
              await deleteDomainRelationship(
                relationshipId
              );

            const data =
              await response.json();

            if (
              !response.ok ||
              !data.success
            ) {
              throw new Error(
                data.error ||
                  "Unable to remove a metadata domain relationship."
              );
            }
          }
        }

        /*
         * Create newly selected relationships.
         */
        for (
          const relationship of desiredRelationships
        ) {
          const key =
            keyFor(
              relationship
            );

          if (
            currentKeys.has(key)
          ) {
            continue;
          }

          const response =
            await createDomainRelationship({
              metadata_record_id:
                numericMetadataId,
              domain_id:
                relationship.domainId,
              relationship_type:
                relationship.relationshipType,
            });

          const data =
            await response.json();

          if (
            !response.ok ||
            !data.success
          ) {
            throw new Error(
              data.error ||
                "Unable to create a metadata domain relationship."
            );
          }
        }

        /*
         * Re-read the server state so the UI reflects
         * the actual persisted relationships.
         */
        const finalResponse =
          await getMetadataDomainRelationships(
            numericMetadataId
          );

        const finalData =
          await finalResponse.json();

        if (
          !finalResponse.ok ||
          !finalData.success
        ) {
          throw new Error(
            finalData.error ||
              "Domain relationships were saved but could not be reloaded."
          );
        }

        const finalSelections =
          extractRelationshipSelections(
            finalData
          );

        setMetadataDomainRelationships(
          finalSelections
        );

        setAppraisalDomainRelationships(
          finalSelections
        );

        return finalSelections;
      } catch (error) {
        const message =
          error instanceof Error
            ? error.message
            : "Unable to save metadata domain relationships.";

        setDomainRelationshipsError(
          message
        );

        throw error;
      } finally {
        setDomainRelationshipsLoading(
          false
        );
      }
    };

      setLoading(true);

        dashboardResponse,
        metadataResponse,
        departmentsResponse,
        typesResponse,
        documentsResponse,
      ] = await Promise.all([
        getDashboard(),
        getMetadata(),
        getDepartments(),
        getDocumentTypes(),
        getDocuments(),
      ]);

      const [
        dashboardData,
        metadataData,
        departmentsData,
        typesData,
        documentsData,
      ] = await Promise.all([
        dashboardResponse.json(),
        metadataResponse.json(),
        departmentsResponse.json(),
        typesResponse.json(),
        documentsResponse.json(),
      ]);

      if (dashboardData.success) {
        setStatistics(
          dashboardData.statistics ??
            dashboardData.dashboard ??
            null
        );
      }

      if (metadataData.success) {
        setMetadata(
          Array.isArray(
            metadataData.metadata
          )
            ? metadataData.metadata
            : []
        );
      }

      if (departmentsData.success) {
        setDepartments(
          Array.isArray(
            departmentsData.departments
          )
            ? departmentsData.departments
            : []
        );
      }

      if (typesData.success) {
        setDocumentTypes(
          Array.isArray(
            typesData.documentTypes
          )
            ? typesData.documentTypes
            : []
        );
      }

      if (documentsData.success) {
        setDocuments(
          Array.isArray(
            documentsData.documents
          )
            ? documentsData.documents
            : []
        );
      }
    } catch (error) {
      console.error(
        "Unable to load system data:",
        error
      );
    } finally {
      setLoading(false);
    }
  };

const handleMetadataSubmit = async (
    event: FormEvent<HTMLFormElement>
  ) => {
    if (!canManageMetadata) {
      return;
    }

    event.preventDefault();

    try {
      setDomainRelationshipsError("");

      const response =
        await createMetadata({
          referenceCode:
            form.referenceCode.trim(),

          personName:
            form.personName.trim() ||
            null,

          title:
            form.title.trim() ||
            null,

          departmentId:
            form.departmentId ||
            null,

          documentTypeId:
            form.documentTypeId ||
            null,

          year:
            form.year ||
            null,

          documentDate:
            form.documentDate ||
            null,

          section:
            form.section.trim() ||
            null,

          description:
            form.description.trim() ||
            null,
        });

      const data =
        await response.json();

      if (
        !response.ok ||
        !data.success
      ) {
        throw new Error(
          data.error ||
            "Unable to create metadata record."
        );
      }

      const metadataRecordId =
        Number(
          data.metadata?.id ??
            data.record?.id ??
            data.id ??
            0
        );

      if (
        Number.isInteger(
          metadataRecordId
        ) &&
        metadataRecordId > 0
      ) {
        await saveMetadataDomainRelationships(
          metadataRecordId,
          metadataDomainRelationships
        );
      }

      setForm({
        ...EMPTY_FORM,
      });

      setMetadataDomainRelationships(
        []
      );

      closeMetadataModal();

      await loadData();
    } catch (error) {
      window.alert(
        error instanceof Error
          ? error.message
          : "Unable to create metadata record."
      );
    }
  };

      event.target.files?.[0] ??
      null;

    setUploadError("");
    setUploadedDocument(null);
    setUploadedAppraisal(null);
    setAppraisalDomainRelationships(
      []
    );

    if (!file) {
      setSelectedFile(null);
      return;
    }

    const allowedTypes = [
      "application/pdf",
      "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
    ];

    const extension =
      file.name
        .toLowerCase()
        .split(".")
        .pop();

    if (
      !allowedTypes.includes(
        file.type
      ) &&
      extension !== "pdf" &&
      extension !== "docx"
    ) {
      setUploadError(
        "Only PDF and DOCX documents are currently supported."
      );

      setSelectedFile(null);
      event.target.value = "";
      return;
    }

    setSelectedFile(file);
  };

const handleUpload = async (
    event: FormEvent<HTMLFormElement>
  ) => {
    if (!canManageDocuments) {
      return;
    }

    event.preventDefault();

    if (!selectedFile) {
      setUploadError(
        "Please select a PDF or DOCX document."
      );
      return;
    }

    try {
      setUploading(true);
      setUploadError("");
      setUploadedDocument(null);
      setUploadedAppraisal(null);
      setReviewingDecision(false);
      setReviewError("");
      setAppraisalDomainRelationships(
        []
      );

      const response =
        await uploadDocument(
          selectedFile
        );

      const data =
        await response.json();

      if (
        !response.ok ||
        !data.success
      ) {
        throw new Error(
          data.error ||
            "Document upload failed."
        );
      }

      if (
        !data.document ||
        typeof data.document !==
          "object"
      ) {
        throw new Error(
          "Document uploaded, but the API returned an invalid document response."
        );
      }

      setUploadedDocument(
        data.document as UploadedDocument
      );

      if (
        data.appraisal &&
        typeof data.appraisal ===
          "object"
      ) {
        const appraisal =
          data.appraisal as UploadedAppraisal;

        setUploadedAppraisal(
          appraisal
        );

        const metadata =
          appraisal.metadata ??
          null;

        setAppraisalMetadataForm(
          {
            referenceCode:
              metadata?.reference_code ??
              data.metadata?.reference_code ??
              "",

            title:
              metadata?.title ??
              data.metadata?.title ??
              "",

            documentDate:
              metadata?.document_date ??
              data.metadata?.document_date ??
              "",

            year:
              metadata?.year != null
                ? String(
                    metadata.year
                  )
                : data.metadata
                      ?.year != null
                  ? String(
                      data.metadata
                        .year
                    )
                  : "",

            personName:
              metadata?.person_name ??
              data.metadata?.person_name ??
              "",

            departmentId:
              metadata?.department_id !=
              null
                ? String(
                    metadata.department_id
                  )
                : "",

            documentTypeId:
              metadata?.document_type_id !=
              null
                ? String(
                    metadata.document_type_id
                  )
                : "",

            section:
              metadata?.section ??
              data.metadata?.section ??
              "",

            description:
              metadata?.description ??
              data.metadata?.description ??
              "",
          }
        );

        setAppraisalCommittedAdditionalMetadata(
          Object.entries(
            metadata?.additional_metadata ??
              data.metadata
                ?.additional_metadata ??
              {}
          ).map(
            ([key, value]) => ({
              key,
              value:
                value === null ||
                value ===
                  undefined
                  ? ""
                  : String(value),
            })
          )
        );

        setAppraisalAdditionalMetadata(
          []
        );

        const metadataRecordId =
          appraisal.metadataRecordId ??
          data.metadata?.id ??
          null;

        if (
          metadataRecordId
        ) {
          await loadMetadataDomainRelationships(
            metadataRecordId
          );
        } else {
          setAppraisalDomainRelationships(
            []
          );
        }
      } else {
        setUploadedAppraisal(
          null
        );

        setAppraisalDomainRelationships(
          []
        );
      }

      setReviewError("");
      setSelectedFile(null);

      const fileInput =
        document.getElementById(
          "document-upload"
        ) as HTMLInputElement | null;

      if (fileInput) {
        fileInput.value = "";
      }

      await loadData();
    } catch (error) {
      setUploadError(
        error instanceof Error
          ? error.message
          : "Unable to upload document."
      );
    } finally {
      setUploading(false);
    }
  };

const handleOpenDocument =
    async (
      documentId:
        | string
        | number
    ) => {
      const numericDocumentId =
        Number(documentId);

      if (
        !Number.isInteger(
          numericDocumentId
        ) ||
        numericDocumentId <= 0
      ) {
        return;
      }

      const documentWindow =
        window.open(
          "",
          "_blank"
        );

      if (!documentWindow) {
        setReviewError(
          "Unable to open the document. Please allow pop-ups for this site and try again."
        );

        return;
      }

      documentWindow.document.title =
        "Opening document...";

      try {
        setReviewError("");

        const response =
          await getDocumentFile(
            numericDocumentId
          );

        if (!response.ok) {
          let errorMessage =
            "Unable to open the document.";

          try {
            const data =
              await response.json();

            if (
              data &&
              typeof data.error ===
                "string" &&
              data.error.trim()
            ) {
              errorMessage =
                data.error;
            }
          } catch {
            // Non-JSON error response.
          }

          throw new Error(
            errorMessage
          );
        }

        const blob =
          await response.blob();

        const objectUrl =
          URL.createObjectURL(
            blob
          );

        documentWindow.location.href =
          objectUrl;

        window.setTimeout(
          () => {
            URL.revokeObjectURL(
              objectUrl
            );
          },
          60_000
        );
      } catch (error) {
        documentWindow.close();

        console.error(
          "Document opening failed:",
          error
        );

        setReviewError(
          error instanceof Error
            ? error.message
            : "Unable to open the document."
        );
      }
    };

const normalizeAppraisalDocumentDate =
    (
      value: unknown
    ): string => {
      const raw =
        String(
          value ?? ""
        ).trim();

      if (
        /^\d{4}-\d{2}-\d{2}$/.test(
          raw
        )
      ) {
        return raw;
      }

      return "";
    };

const buildAppraisalAdditionalMetadata =
    (): Record<
      string,
      unknown
    > => {
      const entries = [
        ...appraisalCommittedAdditionalMetadata,
        ...appraisalAdditionalMetadata,
      ]
        .map(
          (field) => ({
            key: String(
              field.key ?? ""
            ).trim(),

            value: String(
              field.value ?? ""
            ).trim(),
          })
        )
        .filter(
          (field) =>
            field.key.length >
              0 &&
            field.value.length >
              0
        );

      return entries.reduce<
        Record<
          string,
          unknown
        >
      >(
        (
          result,
          field
        ) => {
          result[field.key] =
            field.value;

          return result;
        },
        {}
      );
    };

const handleDoneAppraisal =
    async () => {
      if (!canManageDocuments) {
        return;
      }

      if (
        !uploadedDocument?.id ||
        !uploadedAppraisal
      ) {
        return;
      }

      if (
        isCreatingNewMetadata
      ) {
        closeUploadModal();
        return;
      }

      const metadataRecordId =
        uploadedAppraisal.metadataRecordId ??
        null;

      const referenceCode =
        appraisalMetadataForm.referenceCode.trim() ||
        String(
          uploadedAppraisal.metadata
            ?.reference_code ??
            ""
        ).trim();

      if (
        !metadataRecordId &&
        !referenceCode
      ) {
        setReviewError(
          "A metadata reference code has not been assigned to this document."
        );

        return;
      }

      try {
        setReviewingDecision(true);
        setReviewError("");

        const response =
          await reviewDocument(
            Number(
              uploadedDocument.id
            ),
            {
              decision:
                metadataRecordId !==
                null
                  ? "APPROVE"
                  : "CREATE_METADATA",

              metadataRecordId,

              metadata:
                buildAppraisalMetadata(
                  referenceCode
                ),
            }
          );

        const data =
          await response.json();

        if (
          !response.ok ||
          !data.success
        ) {
          throw new Error(
            data.error ||
              data.message ||
              data.details ||
              "Unable to save the appraisal."
          );
        }

        if (
          data.document
        ) {
          setUploadedDocument(
            data.document
          );
        }

        const persistedMetadataId =
          Number(
            data.metadata?.id ??
              metadataRecordId ??
              0
          );

        if (
          Number.isInteger(
            persistedMetadataId
          ) &&
          persistedMetadataId >
            0
        ) {
          await saveMetadataDomainRelationships(
            persistedMetadataId,
            appraisalDomainRelationships
          );
        }

        await loadData();

        setReviewError("");
        closeUploadModal();
      } catch (error) {
        setReviewError(
          error instanceof Error
            ? error.message
            : "Unable to save the appraisal."
        );
      } finally {
        setReviewingDecision(
          false
        );
      }
    };
