const saveMetadataDomainRelationships =
    async (
      metadataRecordId: string | number,
      desiredRelationships:
        DomainRelationshipSelection[]
    ) => {
      const numericMetadataId =
        Number(metadataRecordId);

      if (
        !Number.isInteger(
          numericMetadataId
        ) ||
        numericMetadataId <= 0
      ) {
        throw new Error(
          "A valid metadata record is required before domain relationships can be saved."
        );
      }

      setDomainRelationshipsLoading(
        true
      );
      setDomainRelationshipsError("");

      try {
        const currentResponse =
          await getMetadataDomainRelationships(
            numericMetadataId
          );

        const currentData =
          await currentResponse.json();

        if (
          !currentResponse.ok ||
          !currentData.success
        ) {
          throw new Error(
            currentData.error ||
              "Unable to retrieve the current metadata domain relationships."
          );
        }

        const currentRelationships =
          Array.isArray(
            currentData.relationships
          )
            ? currentData.relationships
            : [];

        const currentSelections =
          extractRelationshipSelections(
            currentData
          );

        const keyFor =
          (
            relationship: DomainRelationshipSelection
          ) =>
            `${relationship.domainId}:${relationship.relationshipType}`;

        const desiredKeys =
          new Set(
            desiredRelationships.map(
              keyFor
            )
          );

        const currentKeys =
          new Set(
            currentSelections.map(
              keyFor
            )
          );

        /*
         * Delete relationships that were removed
         * from the selector.
         */
        for (
          const relationship of currentRelationships
        ) {
          if (
            !relationship ||
            typeof relationship !==
              "object"
          ) {
            continue;
          }

          const item =
            relationship as Record<
              string,
              unknown
            >;

          const relationshipId =
            Number(
              item.id
            );

          const domainId =
            Number(
              item.domain_id ??
                item.domainId
            );

          const relationshipType =
            String(
              item.relationship_type ??
                item.relationshipType ??
                ""
            ).trim();

          if (
            !Number.isInteger(
              relationshipId
            ) ||
            relationshipId <= 0 ||
            !Number.isInteger(
              domainId
            ) ||
            domainId <= 0 ||
            !relationshipType
          ) {
            continue;
          }

          const key =
            `${domainId}:${relationshipType}`;

          if (
            !desiredKeys.has(key)
          ) {
            const response =
              await deleteDomainRelationship(
                relationshipId
              );

            const data =
              await response.json();

            if (
              !response.ok ||
              !data.success
            ) {
              throw new Error(
                data.error ||
                  "Unable to remove a metadata domain relationship."
              );
            }
          }
        }

        /*
         * Create newly selected relationships.
         */
        for (
          const relationship of desiredRelationships
        ) {
          const key =
            keyFor(
              relationship
            );

          if (
            currentKeys.has(key)
          ) {
            continue;
          }

          const response =
            await createDomainRelationship({
              metadata_record_id:
                numericMetadataId,
              domain_id:
                relationship.domainId,
              relationship_type:
                relationship.relationshipType,
            });

          const data =
            await response.json();

          if (
            !response.ok ||
            !data.success
          ) {
            throw new Error(
              data.error ||
                "Unable to create a metadata domain relationship."
            );
          }
        }

        /*
         * Re-read the server state so the UI reflects
         * the actual persisted relationships.
         */
        const finalResponse =
          await getMetadataDomainRelationships(
            numericMetadataId
          );

        const finalData =
          await finalResponse.json();

        if (
          !finalResponse.ok ||
          !finalData.success
        ) {
          throw new Error(
            finalData.error ||
              "Domain relationships were saved but could not be reloaded."
          );
        }

        const finalSelections =
          extractRelationshipSelections(
            finalData
          );

        setMetadataDomainRelationships(
          finalSelections
        );

        setAppraisalDomainRelationships(
          finalSelections
        );

        return finalSelections;
      } catch (error) {
        const message =
          error instanceof Error
            ? error.message
            : "Unable to save metadata domain relationships.";

        setDomainRelationshipsError(
          message
        );

        throw error;
      } finally {
        setDomainRelationshipsLoading(
          false
        );
      }
    };

const handleFileChange = (
    event: ChangeEvent<HTMLInputElement>
  ) => {
    const file =
      event.target.files?.[0] ??
      null;

    setUploadError("");
    setUploadedDocument(null);
    setUploadedAppraisal(null);
    setAppraisalDomainRelationships(
      []
    );

    if (!file) {
      setSelectedFile(null);
      return;
    }

    const allowedTypes = [
      "application/pdf",
      "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
    ];

    const extension =
      file.name
        .toLowerCase()
        .split(".")
        .pop();

    if (
      !allowedTypes.includes(
        file.type
      ) &&
      extension !== "pdf" &&
      extension !== "docx"
    ) {
      setUploadError(
        "Only PDF and DOCX documents are currently supported."
      );

      setSelectedFile(null);
      event.target.value = "";
      return;
    }

    setSelectedFile(file);
  };

const handleUpload = async (
    event: FormEvent<HTMLFormElement>
  ) => {
    if (!canManageDocuments) {
      return;
    }

    event.preventDefault();

    if (!selectedFile) {
      setUploadError(
        "Please select a PDF or DOCX document."
      );
      return;
    }

    try {
      setUploading(true);
      setUploadError("");
      setUploadedDocument(null);
      setUploadedAppraisal(null);
      setReviewingDecision(false);
      setReviewError("");
      setAppraisalDomainRelationships(
        []
      );

      const response =
        await uploadDocument(
          selectedFile
        );

      const data =
        await response.json();

      if (
        !response.ok ||
        !data.success
      ) {
        throw new Error(
          data.error ||
            "Document upload failed."
        );
      }

      if (
        !data.document ||
        typeof data.document !==
          "object"
      ) {
        throw new Error(
          "Document uploaded, but the API returned an invalid document response."
        );
      }

      setUploadedDocument(
        data.document as UploadedDocument
      );

      if (
        data.appraisal &&
        typeof data.appraisal ===
          "object"
      ) {
        const appraisal =
          data.appraisal as UploadedAppraisal;

        setUploadedAppraisal(
          appraisal
        );

        const metadata =
          appraisal.metadata ??
          null;

        setAppraisalMetadataForm(
          {
            referenceCode:
              metadata?.reference_code ??
              data.metadata?.reference_code ??
              "",

            title:
              metadata?.title ??
              data.metadata?.title ??
              "",

            documentDate:
              metadata?.document_date ??
              data.metadata?.document_date ??
              "",

            year:
              metadata?.year != null
                ? String(
                    metadata.year
                  )
                : data.metadata
                      ?.year != null
                  ? String(
                      data.metadata
                        .year
                    )
                  : "",

            personName:
              metadata?.person_name ??
              data.metadata?.person_name ??
              "",

            departmentId:
              metadata?.department_id !=
              null
                ? String(
                    metadata.department_id
                  )
                : "",

            documentTypeId:
              metadata?.document_type_id !=
              null
                ? String(
                    metadata.document_type_id
                  )
                : "",

            section:
              metadata?.section ??
              data.metadata?.section ??
              "",

            description:
              metadata?.description ??
              data.metadata?.description ??
              "",
          }
        );

        setAppraisalCommittedAdditionalMetadata(
          Object.entries(
            metadata?.additional_metadata ??
              data.metadata
                ?.additional_metadata ??
              {}
          ).map(
            ([key, value]) => ({
              key,
              value:
                value === null ||
                value ===
                  undefined
                  ? ""
                  : String(value),
            })
          )
        );

        setAppraisalAdditionalMetadata(
          []
        );

        const metadataRecordId =
          appraisal.metadataRecordId ??
          data.metadata?.id ??
          null;

        if (
          metadataRecordId
        ) {
          await loadMetadataDomainRelationships(
            metadataRecordId
          );
        } else {
          setAppraisalDomainRelationships(
            []
          );
        }
      } else {
        setUploadedAppraisal(
          null
        );

        setAppraisalDomainRelationships(
          []
        );
      }

      setReviewError("");
      setSelectedFile(null);

      const fileInput =
        document.getElementById(
          "document-upload"
        ) as HTMLInputElement | null;

      if (fileInput) {
        fileInput.value = "";
      }

      await loadData();
    } catch (error) {
      setUploadError(
        error instanceof Error
          ? error.message
          : "Unable to upload document."
      );
    } finally {
      setUploading(false);
    }
  };

const normalizeAppraisalDocumentDate =
    (
      value: unknown
    ): string => {
      const raw =
        String(
          value ?? ""
        ).trim();

      if (
        /^\d{4}-\d{2}-\d{2}$/.test(
          raw
        )
      ) {
        return raw;
      }

      return "";
    };

const buildAppraisalAdditionalMetadata =
    (): Record<
      string,
      unknown
    > => {
      const entries = [
        ...appraisalCommittedAdditionalMetadata,
        ...appraisalAdditionalMetadata,
      ]
        .map(
          (field) => ({
            key: String(
              field.key ?? ""
            ).trim(),

            value: String(
              field.value ?? ""
            ).trim(),
          })
        )
        .filter(
          (field) =>
            field.key.length >
              0 &&
            field.value.length >
              0
        );

      return entries.reduce<
        Record<
          string,
          unknown
        >
      >(
        (
          result,
          field
        ) => {
          result[field.key] =
            field.value;

          return result;
        },
        {}
      );
    };

const handleDoneAppraisal =
    async () => {
      if (!canManageDocuments) {
        return;
      }

      if (
        !uploadedDocument?.id ||
        !uploadedAppraisal
      ) {
        return;
      }

      if (
        isCreatingNewMetadata
      ) {
        closeUploadModal();
        return;
      }

      const metadataRecordId =
        uploadedAppraisal.metadataRecordId ??
        null;

      const referenceCode =
        appraisalMetadataForm.referenceCode.trim() ||
        String(
          uploadedAppraisal.metadata
            ?.reference_code ??
            ""
        ).trim();

      if (
        !metadataRecordId &&
        !referenceCode
      ) {
        setReviewError(
          "A metadata reference code has not been assigned to this document."
        );

        return;
      }

      try {
        setReviewingDecision(true);
        setReviewError("");

        const response =
          await reviewDocument(
            Number(
              uploadedDocument.id
            ),
            {
              decision:
                metadataRecordId !==
                null
                  ? "APPROVE"
                  : "CREATE_METADATA",

              metadataRecordId,

              metadata:
                buildAppraisalMetadata(
                  referenceCode
                ),
            }
          );

        const data =
          await response.json();

        if (
          !response.ok ||
          !data.success
        ) {
          throw new Error(
            data.error ||
              data.message ||
              data.details ||
              "Unable to save the appraisal."
          );
        }

        if (
          data.document
        ) {
          setUploadedDocument(
            data.document
          );
        }

        const persistedMetadataId =
          Number(
            data.metadata?.id ??
              metadataRecordId ??
              0
          );

        if (
          Number.isInteger(
            persistedMetadataId
          ) &&
          persistedMetadataId >
            0
        ) {
          await saveMetadataDomainRelationships(
            persistedMetadataId,
            appraisalDomainRelationships
          );
        }

        await loadData();

        setReviewError("");
        closeUploadModal();
      } catch (error) {
        setReviewError(
          error instanceof Error
            ? error.message
            : "Unable to save the appraisal."
        );
      } finally {
        setReviewingDecision(
          false
        );
      }
    };

const handleAppraisalReview =
    async (
      decision:
        | "APPROVE"
        | "REJECT"
    ) => {
      if (!canManageDocuments) {
        return;
      }

      if (
        !uploadedDocument?.id ||
        !uploadedAppraisal
      ) {
        return;
      }

      if (
        decision === "REJECT"
      ) {
        setReviewError("");
        setIsCreatingNewMetadata(
          true
        );

        return;
      }

      if (
        !uploadedAppraisal.metadataRecordId
      ) {
        setReviewError(
          "There is no metadata record available to approve."
        );

        return;
      }

      try {
        setReviewingDecision(
          true
        );

        setReviewError("");

        const response =
          await reviewDocument(
            Number(
              uploadedDocument.id
            ),
            {
              decision:
                "APPROVE",

              metadataRecordId:
                uploadedAppraisal.metadataRecordId,

              metadata:
                buildAppraisalMetadata(
                  appraisalMetadataForm.referenceCode.trim()
                ),
            }
          );

        const data =
          await response.json();

        if (
          !response.ok ||
          !data.success
        ) {
          throw new Error(
            data.error ||
              data.message ||
              data.details ||
              "Unable to approve the appraisal decision."
          );
        }

        if (
          data.document
        ) {
          setUploadedDocument(
            data.document
          );
        }

        const persistedMetadataId =
          Number(
            data.metadata?.id ??
              uploadedAppraisal.metadataRecordId ??
              0
          );

        if (
          Number.isInteger(
            persistedMetadataId
          ) &&
          persistedMetadataId >
            0
        ) {
          await saveMetadataDomainRelationships(
            persistedMetadataId,
            appraisalDomainRelationships
          );
        }

        await loadData();

        setUploadedAppraisal(
          (
            current
          ) =>
            current
              ? {
                  ...current,
                  decision:
                    "LINKED",
                  metadataRecordId:
                    persistedMetadataId ||
                    current.metadataRecordId,
                }
              : current
        );

        setReviewError("");
      } catch (error) {
        setReviewError(
          error instanceof Error
            ? error.message
            : "Unable to approve the appraisal decision."
        );
      } finally {
        setReviewingDecision(
          false
        );
      }
    };
