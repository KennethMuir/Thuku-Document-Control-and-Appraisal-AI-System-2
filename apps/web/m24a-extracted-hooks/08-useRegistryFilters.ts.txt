const loadDomainRelationshipDefinitions =
    async () => {
      if (!authUser) {
        return;
      }

      try {
        setDomainRelationshipsLoading(true);
        setDomainRelationshipsError("");

        const [
          domainsResponse,
          relationshipTypesResponse,
        ] = await Promise.all([
          getDomainRegistry(),
          getRelationshipTypes(),
        ]);

        const [
          domainsData,
          relationshipTypesData,
        ] = await Promise.all([
          domainsResponse.json(),
          relationshipTypesResponse.json(),
        ]);

        if (
          !domainsResponse.ok ||
          !domainsData.success
        ) {
          throw new Error(
            domainsData.error ||
              "Unable to load the domain registry."
          );
        }

        if (
          !relationshipTypesResponse.ok ||
          !relationshipTypesData.success
        ) {
          throw new Error(
            relationshipTypesData.error ||
              "Unable to load domain relationship types."
          );
        }

        const rawDomains =
          Array.isArray(domainsData.domains)
            ? domainsData.domains
            : [];

        const normalizedDomains =
          rawDomains
            .map(
              (
                domain: Record<string, unknown>
              ) => {
                const id = Number(
                  domain.id
                );

                const domainKey =
                  String(
                    domain.domain_key ??
                      domain.domainKey ??
                      ""
                  ).trim();

                const displayName =
                  String(
                    domain.display_name ??
                      domain.displayName ??
                      domainKey
                  ).trim();

                if (
                  !Number.isInteger(id) ||
                  id <= 0 ||
                  !domainKey
                ) {
                  return null;
                }

                return {
                  id,
                  domain_key: domainKey,
                  display_name:
                    displayName,
                  description:
                    domain.description == null
                      ? null
                      : String(
                          domain.description
                        ),
                  domain_family:
                    domain.domain_family == null
                      ? null
                      : String(
                          domain.domain_family
                        ),
                };
              }
            )
            .filter(
              (domain: unknown): domain is DomainRelationshipDomain =>
                domain !== null
            );

        const rawTypes =
          Array.isArray(
            relationshipTypesData.relationshipTypes
          )
            ? relationshipTypesData.relationshipTypes
            : Array.isArray(
                relationshipTypesData.types
              )
              ? relationshipTypesData.types
              : [];

        const normalizedTypes =
          rawTypes
            .map(
              (
                relationshipType: unknown
              ) => {
                if (
                  typeof relationshipType ===
                  "string"
                ) {
                  return {
                    value:
                      relationshipType,
                    label:
                      relationshipType,
                  };
                }

                if (
                  !relationshipType ||
                  typeof relationshipType !==
                    "object"
                ) {
                  return null;
                }

                const item =
                  relationshipType as Record<
                    string,
                    unknown
                  >;

                const value =
                  String(
                    item.value ??
                      item.relationship_type ??
                      item.relationshipType ??
                      item.key ??
                      ""
                  ).trim();

                if (!value) {
                  return null;
                }

                return {
                  value,
                  label:
                    String(
                      item.label ??
                        item.display_name ??
                        item.displayName ??
                        value
                    ).trim() || value,
                };
              }
            )
            .filter(
              (relationshipType: unknown): relationshipType is DomainRelationshipType =>
                relationshipType !== null
            );

        setDomainRelationshipDomains(
          normalizedDomains
        );

        setDomainRelationshipTypes(
          normalizedTypes
        );
      } catch (error) {
        console.error(
          "Unable to load domain relationship definitions:",
          error
        );

        setDomainRelationshipsError(
          error instanceof Error
            ? error.message
            : "Unable to load domain relationships."
        );
      } finally {
        setDomainRelationshipsLoading(
          false
        );
      }
    };

const extractRelationshipSelections =
    (
      data: Record<string, unknown>
    ): DomainRelationshipSelection[] => {
      const relationships =
        Array.isArray(
          data.relationships
        )
          ? data.relationships
          : [];

      return relationships
        .map(
          (
            relationship: unknown
          ) => {
            if (
              !relationship ||
              typeof relationship !==
                "object"
            ) {
              return null;
            }

            const item =
              relationship as Record<
                string,
                unknown
              >;

            const domainId = Number(
              item.domain_id ??
                item.domainId ??
                (
                  item.domain &&
                  typeof item.domain ===
                    "object"
                    ? (
                        item.domain as Record<
                          string,
                          unknown
                        >
                      ).id
                    : undefined
                )
            );

            const relationshipType =
              String(
                item.relationship_type ??
                  item.relationshipType ??
                  ""
              ).trim();

            if (
              !Number.isInteger(
                domainId
              ) ||
              domainId <= 0 ||
              !relationshipType
            ) {
              return null;
            }

            return {
              domainId,
              relationshipType,
            };
          }
        )
        .filter(
          (
            relationship
          ): relationship is DomainRelationshipSelection =>
            relationship !== null
        );
    };

const buildAppraisalAdditionalMetadata =
    (): Record<
      string,
      unknown
    > => {
      const entries = [
        ...appraisalCommittedAdditionalMetadata,
        ...appraisalAdditionalMetadata,
      ]
        .map(
          (field) => ({
            key: String(
              field.key ?? ""
            ).trim(),

            value: String(
              field.value ?? ""
            ).trim(),
          })
        )
        .filter(
          (field) =>
            field.key.length >
              0 &&
            field.value.length >
              0
        );

      return entries.reduce<
        Record<
          string,
          unknown
        >
      >(
        (
          result,
          field
        ) => {
          result[field.key] =
            field.value;

          return result;
        },
        {}
      );
    };

const filteredMetadata =
    metadata.filter(
      (
        record: MetadataRecord
      ) => {
        if (
          dashboardMetadataFilter !==
            "ALL" &&
          record.status !==
            dashboardMetadataFilter
        ) {
          return false;
        }

        if (
          selectedDepartmentId !==
          null
        ) {
          const department =
            departments.find(
              (
                item
              ) =>
                item.id ===
                selectedDepartmentId
            );

          if (
            !department ||
            record.department_name !==
              department.name
          ) {
            return false;
          }
        }

        const query =
          searchTerm
            .trim()
            .toLowerCase();

        if (!query) {
          return true;
        }

        return [
          record.reference_code,
          record.title,
          record.person_name,
          record.description,
          record.section,
          record.department_name,
          record.document_type_name,
          record.status,
          record.linked_document_code,
          record.linked_filename,
        ]
          .filter(Boolean)
          .some(
            (
              value
            ) =>
              String(
                value
              )
                .toLowerCase()
                .includes(
                  query
                )
          );
      }
    );
