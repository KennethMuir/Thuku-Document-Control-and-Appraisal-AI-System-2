const loadDomainRelationshipDefinitions =
    async () => {
      if (!authUser) {
        return;
      }

      try {
        setDomainRelationshipsLoading(true);
        setDomainRelationshipsError("");

        const [
          domainsResponse,
          relationshipTypesResponse,
        ] = await Promise.all([
          getDomainRegistry(),
          getRelationshipTypes(),
        ]);

        const [
          domainsData,
          relationshipTypesData,
        ] = await Promise.all([
          domainsResponse.json(),
          relationshipTypesResponse.json(),
        ]);

        if (
          !domainsResponse.ok ||
          !domainsData.success
        ) {
          throw new Error(
            domainsData.error ||
              "Unable to load the domain registry."
          );
        }

        if (
          !relationshipTypesResponse.ok ||
          !relationshipTypesData.success
        ) {
          throw new Error(
            relationshipTypesData.error ||
              "Unable to load domain relationship types."
          );
        }

        const rawDomains =
          Array.isArray(domainsData.domains)
            ? domainsData.domains
            : [];

        const normalizedDomains =
          rawDomains
            .map(
              (
                domain: Record<string, unknown>
              ) => {
                const id = Number(
                  domain.id
                );

                const domainKey =
                  String(
                    domain.domain_key ??
                      domain.domainKey ??
                      ""
                  ).trim();

                const displayName =
                  String(
                    domain.display_name ??
                      domain.displayName ??
                      domainKey
                  ).trim();

                if (
                  !Number.isInteger(id) ||
                  id <= 0 ||
                  !domainKey
                ) {
                  return null;
                }

                return {
                  id,
                  domain_key: domainKey,
                  display_name:
                    displayName,
                  description:
                    domain.description == null
                      ? null
                      : String(
                          domain.description
                        ),
                  domain_family:
                    domain.domain_family == null
                      ? null
                      : String(
                          domain.domain_family
                        ),
                };
              }
            )
            .filter(
              (domain: unknown): domain is DomainRelationshipDomain =>
                domain !== null
            );

        const rawTypes =
          Array.isArray(
            relationshipTypesData.relationshipTypes
          )
            ? relationshipTypesData.relationshipTypes
            : Array.isArray(
                relationshipTypesData.types
              )
              ? relationshipTypesData.types
              : [];

        const normalizedTypes =
          rawTypes
            .map(
              (
                relationshipType: unknown
              ) => {
                if (
                  typeof relationshipType ===
                  "string"
                ) {
                  return {
                    value:
                      relationshipType,
                    label:
                      relationshipType,
                  };
                }

                if (
                  !relationshipType ||
                  typeof relationshipType !==
                    "object"
                ) {
                  return null;
                }

                const item =
                  relationshipType as Record<
                    string,
                    unknown
                  >;

                const value =
                  String(
                    item.value ??
                      item.relationship_type ??
                      item.relationshipType ??
                      item.key ??
                      ""
                  ).trim();

                if (!value) {
                  return null;
                }

                return {
                  value,
                  label:
                    String(
                      item.label ??
                        item.display_name ??
                        item.displayName ??
                        value
                    ).trim() || value,
                };
              }
            )
            .filter(
              (relationshipType: unknown): relationshipType is DomainRelationshipType =>
                relationshipType !== null
            );

        setDomainRelationshipDomains(
          normalizedDomains
        );

        setDomainRelationshipTypes(
          normalizedTypes
        );
      } catch (error) {
        console.error(
          "Unable to load domain relationship definitions:",
          error
        );

        setDomainRelationshipsError(
          error instanceof Error
            ? error.message
            : "Unable to load domain relationships."
        );
      } finally {
        setDomainRelationshipsLoading(
          false
        );
      }
    };

useEffect(() => {
    let cancelled = false;

    const verifySession =
      async () => {
        try {
          setAuthError("");

          const response =
            await getCurrentUser();

          const data =
            await response.json();

          if (cancelled) {
            return;
          }

          if (
            !response.ok ||
            !data.success ||
            !data.authenticated ||
            !data.user
          ) {
            setAuthUser(null);

            setAuthError(
              data.error ||
                "Your session has expired. Please sign in again."
            );

            return;
          }

          setAuthUser(data.user);
        } catch (error) {
          if (cancelled) {
            return;
          }

          console.error(
            "Session verification failed:",
            error
          );

          setAuthUser(null);

          setAuthError(
            "Unable to verify your session. Please sign in again."
          );
        } finally {
          if (!cancelled) {
            setAuthLoading(false);
          }
        }
      };

    void verifySession();

    return () => {
      cancelled = true;
    };
  }, []);

            const response =
              await searchDocuments(
                query
              );

            const data =
              await response.json();

            if (cancelled) {
              return;
            }

            if (
              !response.ok ||
              !data.success
            ) {
              throw new Error(
                data.error ||
                  "Unable to search records."
              );
            }

            setSearchResults(
              Array.isArray(
                data.results
              )
                ? data.results
                : []
            );
          } catch (error) {
            if (cancelled) {
              return;
            }

            console.error(
              "Search request failed:",
              error
            );

            setSearchResults([]);

            setSearchError(
              error instanceof Error
                ? error.message
                : "Unable to search records."
            );
          } finally {
            if (!cancelled) {
              setSearchLoading(
                false
              );
            }
          }
        },
        300
      );

    return () => {
      cancelled = true;
      window.clearTimeout(timer);
    };
  }, [
    searchTerm,
    authLoading,
    authUser,
  ]);
